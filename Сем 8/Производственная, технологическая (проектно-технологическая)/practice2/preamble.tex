\newcommand{\structuralsection}[1]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{section}{#1}%
  \begin{center}
    \textbf{\MakeUppercase{#1}}
  \end{center}
}

\newcommand{\appendixsection}[2]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{section}{Приложение #1. #2}%
  \begin{flushright}
    \textbf{Приложение #1}
  \end{flushright}
  \begin{center}
    \textbf{#2}
  \end{center}
}
